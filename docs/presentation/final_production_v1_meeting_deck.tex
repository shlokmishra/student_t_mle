\section{Problem and Setup}

\begin{frame}{The problem}
  \claim{A released MLE is a compressed statistic. I want to understand the Bayesian posterior and the latent datasets still compatible with that release.}
  \[
    p(\mu \mid x_{1:n}) \quad \text{versus} \quad p(\mu \mid \hat{\mu}=\mu^\star)
  \]
  \begin{itemize}
    \item If the statistic is sufficient, little is lost.
    \item If not, the release may distort posterior uncertainty.
    \item The same release can also change beliefs about hidden latent extremes.
  \end{itemize}
\end{frame}

\begin{frame}{Models and regimes}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Main evidence regimes}
        \begin{itemize}
          \item Logistic: $n=10,20,50$
          \item Student: $k=2,3$, validated $n$ regimes
          \item Laplace: odd $n=11,21,51$, Gibbs only
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Separated / caveat regimes}
        \begin{itemize}
          \item Student $k=1$: heavy-tail stress test
          \item Student $k=1,n=10$: unresolved
          \item Laplace RATTLE: not applicable because the score/median constraint is non-smooth
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Frozen evidence source}
  \claim{\texttt{final\_production\_v1} is the meeting dataset.}
  \begin{itemize}
    \item 81 completed sampler case directories
    \item 100k iterations, 20k burn-in
    \item 3 seeds per production case
    \item 129,681 thinned latent geometry states
    \item 1,800 paired release-information datasets
  \end{itemize}
\end{frame}

\section{Sampler Correctness}

\begin{frame}{Correctness standard}
  \claim{A sampler passes only when the posterior summaries, invariance checks, and mixing diagnostics tell the same story.}
  \vspace{0.3cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{Posterior agreement}
        Mean, SD, central and tail quantiles, central mass, Wasserstein distance versus the raw weighted-MC benchmark.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Invariant target}
        Constraint residuals, target-mismatch checks, and method-specific move identities.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Chain reliability}
        ESS/autocorrelation, split-chain drift, multiseed stability, and initialization sensitivity when cached.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Method-specific checks}
  \claim{The diagnostics were matched to how each sampler is supposed to preserve the constrained posterior.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \begin{block}{Gibbs}
        \begin{itemize}
          \item Score-constraint residual after sweeps.
          \item Pair update preserves the pairwise $\delta$ identity.
          \item Student inverse-map branch usage is not one-sided.
          \item Seeds, splits, ESS, and latent summaries do not drift.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{RATTLE}
        \begin{itemize}
          \item Projection and tangent residuals stay near numerical zero.
          \item Reverse checks do not fail.
          \item Hamiltonian error is controlled.
          \item Gram/target correction and not-applicable cases are labeled explicitly.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Clean, caveat, unresolved}
  \claim{The label is a presentation decision: what can be used as evidence in the main comparison?}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.32\textwidth}

    \end{column}
    \begin{column}{0.32\textwidth}

\begin{frame}{Why the local RATTLE curve stays flat for a while}
  \claim{The slow observed growth is mostly a systems effect: fixed per-iteration overhead dominates until $n$ gets larger.}
  \begin{itemize}
    \item Each iteration pays substantial nearly-fixed costs: PRNG handling, leapfrog/projection scaffolding, reverse-check bookkeeping, and dispatch/synchronization overhead.
    \item The coordinatewise gradient and projection work is fused into a small number of dense vector kernels, so doubling $n$ does not double wall time immediately.
    \item At these local sizes, the kernels stay in a favorable memory/cache regime; the linear-in-$n$ arithmetic is present but not yet the dominant bottleneck.
    \item The right interpretation is therefore ``sublinear over this measured range,'' not ``RATTLE avoids dimension dependence.''
  \end{itemize}
\end{frame}

\section{Geometry Explanation}

\begin{frame}{Why geometry matters}
  \claim{The latent constraint creates a manifold. The winner is determined by how each sampler moves on it.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \begin{block}{Gibbs}
        Exploits algebra locally through score-preserving pair updates.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{RATTLE}
        Makes geometric moves but pays projection, tangent, and Hamiltonian costs.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{How latent tail classes are defined}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \includegraphics[width=\linewidth]{student_tail_class_definition.png}
    \end{column}
    \begin{column}{0.44\textwidth}
      \claim{These are \textbf{geometry-tail} thresholds: $|y|=\sqrt{k}$ is where $|\psi(y)|$ is largest, not a classical outlier cutoff.}
      \[
        \psi(y)=\frac{y}{k+y^2}
      \]
      \begin{itemize}\setlength{\itemsep}{0.08em}
        \item \textbf{central:} no $|y_i|>\sqrt{k}$
        \item \textbf{mixed tail:} some, but fewer than half
        \item \textbf{tail dominated:} at least half
        \item \textbf{extreme tail:} at least one $|y_i|>4\sqrt{k}$
      \end{itemize}
    \end{column}
  \end{columns}
  \claim{Interpretation: the same threshold means "tail" much earlier for Cauchy than for $t_2$ or $t_3$. These are heuristic latent-geometry labels, not formal posterior modes.}
\end{frame}

\begin{frame}{Percentile interpretation of geometry tails}
  \claim{The score-geometry thresholds are model-dependent. They do not mean equally rare outliers across $k$.}
  \begin{center}
    \begin{tabular}{rcc}
      \toprule
      $k$ & $P(|Y|\le \sqrt{k})$ & $P(|Y|\le 4\sqrt{k})$ \\
      \midrule
      1 & $0.50$ & $0.844$ \\
      2 & $0.71$ & $0.970$ \\
      3 & $0.82$ & $0.994$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \vspace{0.15cm}
  \begin{itemize}
    \item For Cauchy, the geometry-tail threshold starts around the median of $|Y|$.
    \item For $t_2$ and $t_3$, the same rule is much further into the distribution.
    \item So later class-based plots should be read as \textbf{geometry diagnostics}, not classical outlier counts.
  \end{itemize}
\end{frame}

\begin{frame}{Cauchy follow-up: what I ran}
  \claim{I ran a dedicated Student $k=1,n=50$ Gibbs geometry audit rather than treating Cauchy as only a generic caveat.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Run design}
        \begin{itemize}
          \item 15 chains: 5 seeds $\times$ 3 initializations
          \item central, tail-heavy, and random starts
          \item 500k iterations, 100k burn-in
          \item 300{,}015 full latent snapshots analyzed
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Posterior stability}
        \begin{itemize}
          \item mean chain $\mu$: $-1.5\times 10^{-4}$
          \item mean posterior SD: $0.211$
          \item SD range: $0.210$--$0.213$
          \item max $|\bar{\mu}_{chain}|$: about $0.001$
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \figsource{\texttt{results/student\_k1\_n50\_geometry\_audit}}
\end{frame}

\begin{frame}{Cauchy geometry: what we learned}
  \claim{The $\mu$ marginal is stable; the latent worlds remain genuinely heavy-tailed.}
  \begin{itemize}
    \item Any $|y_i|>20$ appears in about $80.9\%$ of thinned states.
    \item Every chain saw a latent coordinate above $2.6\times 10^5$; the maximum was $4.6\times 10^6$.
    \item Mean number of coordinates with $|y_i|>20$ is about $1.63$ out of 50.
    \item About $50.7\%$ of coordinates are on the Student tail branch $|y_i|>1$.
    \item Gibbs branch switching is balanced at about $0.500$.
    \item Tail size is effectively decoupled from $\mu$: $\mathrm{corr}(\mu,\max_i |y_i|)\approx -0.001$.
  \end{itemize}
  \vspace{0.15cm}
  \claim{The caveat is geometric, not posterior instability: huge $|y|$ values are common, but they collapse into bounded score coordinates $z=y/(1+y^2)$.}
\end{frame}

\section{MLE Release Payoff}

\begin{frame}{Final payoff question}
  \claim{Once the sampler is trusted, I ask what the released MLE preserves and what it leaks about hidden data.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \begin{block}{Information loss}
        Compare $p(\mu \mid x_{1:n})$ to $p(\mu \mid \hat{\mu})$ over simulated datasets.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Privacy leakage}
        Compare prior vs posterior beliefs about latent outliers compatible with $\hat{\mu}$.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Sanity check: sufficient statistic}
  \claim{Normal known-variance gives the zero-loss reference.}
  \begin{center}
    \Large
    $p(\mu \mid \bar{x}) = p(\mu \mid x_{1:n})$
  \end{center}
  \begin{itemize}
    \item Median SD ratio: 1.000.
    \item Wasserstein and quantile differences: effectively zero.
    \item This anchors the interpretation of non-normal deviations.
  \end{itemize}
\end{frame}

\section{Discussion}

\begin{frame}{Dashboard walkthrough}
  \claim{Use the dashboard to drill down, but keep the meeting defaults clean.}
  \begin{itemize}
    \item Default source: \texttt{final\_production\_v1}.
    \item Posterior page: Student $k=3,n=20$ or Logistic $n=20$.
    \item KDE page: diagnostic only.
    \item Student $k=1,n=10$: caveat/unresolved.
    \item Laplace RATTLE: hidden or marked not applicable.
  \end{itemize}

\begin{frame}{Caveats}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \begin{block}{Do not headline}
        \begin{itemize}
          \item Student $k=1,n=10$
          \item Student $k=1$ broader heavy-tail claims
          \item Caveat-only small-$n$ rows
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Keep explicit}
        \begin{itemize}
          \item Laplace RATTLE is not applicable
          \item KDE is diagnostic smoothing
          \item Privacy result is belief-shift leakage
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Takeaway}
  \claim{The pipeline now supports a coherent story: validate the sampler, compare efficiency, explain behavior geometrically, then use the trusted sampler to quantify information loss and privacy-relevant leakage.}
  \begin{enumerate}
    \item Clean regimes support Gibbs/RATTLE comparison.
    \item RATTLE is more efficient in smooth comparable regimes.
    \item Heavy tails explain the remaining caveats.
    \item MLE-only release is benign in some regimes and lossy/leaky in others.
  \end{enumerate}
\end{frame}
